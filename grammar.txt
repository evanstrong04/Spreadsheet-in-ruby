expression ::= level0
level0  ::= level1 | level0 "||" level1
level1  ::= level2 | level1 "&&" level2
level2  ::= level3 | level2 "==" level3 | level2 "!=" level3
level3  ::= level4 | level3 "<" level4 | level3 "<=" level4 | level3 ">" level4 | level3 ">=" level4
level4  ::= level5 | level4 "&" level5 | level4 "|" level5 | level4 "^" level5
level5  ::= level6 | level5 "<<" level6 | level5 ">>" level6
level6  ::= level7 | level6 "+" level7 | level6 "-" level7
level7  ::= level8 | level7 "*" level8 | level7 "/" level8 | level7 "%" level8
level8  ::= level9 | level9 "**" level8
level9  ::= level10 | "-" level9 | "!" level9 | "~" level9
level10 ::= integer | float | string | "true" | "false" | "null"
          | "(" expression ")"
          | "#" lvalue | lvalue
          | "int" "(" expression ")" | "float" "(" expression ")"
          | stat "(" lvalue "," lvalue ")"
lvalue  ::= "[" expression "," expression "]"
stat    ::= "sum" | "min" | "max" | "mean"
